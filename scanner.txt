#include <WiFi.h>

// Helper function to translate the raw encryption code into readable text
String translateEncryptionType(wifi_auth_mode_t encryptionType) {
  switch (encryptionType) {
    case (WIFI_AUTH_OPEN): return "Open (UNSECURED)";
    case (WIFI_AUTH_WEP): return "WEP";
    case (WIFI_AUTH_WPA_PSK): return "WPA_PSK";
    case (WIFI_AUTH_WPA2_PSK): return "WPA2_PSK";
    case (WIFI_AUTH_WPA_WPA2_PSK): return "WPA_WPA2_PSK";
    case (WIFI_AUTH_WPA2_ENTERPRISE): return "WPA2_ENTERPRISE";
    case (WIFI_AUTH_WPA3_PSK): return "WPA3_PSK";
    case (WIFI_AUTH_WPA2_WPA3_PSK): return "WPA2_WPA3_PSK";
    default: return "UNKNOWN";
  }
}

void setup() {
  Serial.begin(115200);
  
  // Put ESP32 into Station mode and disconnect from any saved networks
  WiFi.mode(WIFI_STA);
  WiFi.disconnect();
  delay(100);
}

void loop() {
  Serial.println("Starting Detailed Wi-Fi Scan...");
  
  // Perform the scan
  int n = WiFi.scanNetworks();
  
  if (n == 0) {
    Serial.println("No networks found. Is the antenna broken?");
  } else {
    Serial.printf("%d networks found in the area:\n", n);
    Serial.println("==========================================================================================");
    Serial.printf("%-4s | %-25s | %-8s | %-4s | %-17s | %-15s\n", "No.", "SSID (Network Name)", "Signal", "Ch", "MAC Address (BSSID)", "Security");
    Serial.println("------------------------------------------------------------------------------------------");
    
    for (int i = 0; i < n; ++i) {
      // Print detailed information for each network in a table row
      Serial.printf("%-4d | %-25s | %-4d dBm | %-4d | %-17s | %-15s\n", 
                    i + 1, 
                    WiFi.SSID(i).c_str(), 
                    WiFi.RSSI(i), 
                    WiFi.channel(i), 
                    WiFi.BSSIDstr(i).c_str(), 
                    translateEncryptionType(WiFi.encryptionType(i)).c_str());
      delay(10);
    }
    Serial.println("==========================================================================================\n");
  }
  
  // Clear the scan results from memory to prevent memory leaks
  WiFi.scanDelete();
  
  // Wait 10 seconds before running the scan again
  delay(10000);
}
